interface RBI
{
	void display();
}
class UnionBank implements RBI
{
	String statement;
	String accountNo;

	public void display()
	{
		System.out.println("Statement"+statement);
		System.out.println("AccountNo"+accountNo);
	}
}
class StateBank implements RBI
{
	double credit;
	double debit;

	public void display()
	{
		System.out.println("AmountCredited"+credit);
		System.out.println("AmountDebited"+debit);
	}
}
